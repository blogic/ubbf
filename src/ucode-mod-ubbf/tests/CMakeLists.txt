# The sources of ubbf.so as a static library, so a test links the
# functions a module cannot export.
list(TRANSFORM UBBF_LIB_SOURCES PREPEND ${CMAKE_CURRENT_SOURCE_DIR}/../ OUTPUT_VARIABLE core_sources)
add_library(ubbf_test_core STATIC ${core_sources})
target_include_directories(ubbf_test_core SYSTEM PUBLIC ${ucode_include_dir} ${ubox_include_dir} ${nl_tiny_include_dir} ${md_include_dir})
target_include_directories(ubbf_test_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(ubbf_test_core PUBLIC ${ucode_library} ${ubox_library} ${nl_tiny_library} ${md_library} m ${CMAKE_DL_LIBS} ubbf_util)

add_library(ubbf_test_harness STATIC harness.c)
target_include_directories(ubbf_test_harness SYSTEM PUBLIC ${cmocka_include_dir})
target_link_libraries(ubbf_test_harness PUBLIC ubbf_test_core ${cmocka_library} ${json_c_library})

# Each test_<name>.c is one test program. ISOLATED runs it in private
# mount and network namespaces (run-isolated.sh), for code that writes
# /tmp/ubbf or reads /etc/ubbf, such as the instance map, or reads the
# kernel network state.
function(ubbf_unit_test name)
	add_executable(test_${name} test_${name}.c)
	target_link_libraries(test_${name} ubbf_test_harness)
	if("ISOLATED" IN_LIST ARGN)
		add_test(NAME ${name} COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/run-isolated.sh $<TARGET_FILE:test_${name}>)
	else()
		add_test(NAME ${name} COMMAND test_${name})
	endif()
endfunction()
